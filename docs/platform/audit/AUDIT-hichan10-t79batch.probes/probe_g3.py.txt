def _find(o, key, path=""):
    hits = []
    if isinstance(o, dict):
        for k, v in o.items():
            if k == key:
                hits.append((path + "/" + k, v))
            hits += _find(v, key, path + "/" + k)
    elif isinstance(o, list):
        for i, v in enumerate(o):
            hits += _find(v, key, path + "[%d]" % i)
    return hits
