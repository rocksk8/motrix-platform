# ── P2 內部人員（已登入、有效 state）可耗盡「全站」對外查詢額度 ─────────────────
def test_p2_insiders_with_valid_states_can_exhaust_the_global_rate_budget(client, make_user, env, monkeypatch):
    """RED（建議級）：不變式 ⑧ 說『攻擊者不能用垃圾請求把所有人擋在外面』——對**未登入**者成立；
    但 3 個一般帳號各自每 3 秒發起一次（實際節流），60 秒內就能產生 60 個有效 state、各配垃圾 code，
    把全站 60/分 的額度用完，第 4 個人（正常使用者）拿到 429『請稍後再試』。同時每次都真的打了 LINE token 端點。"""
    from helpers import line_login as ll
    monkeypatch.setattr(ll, "START_INTERVAL", 0)             # 以時間快轉代替真等 3 秒：下面用同一個假時鐘
    clock = {"t": 1000.0}
    monkeypatch.setattr(ll, "_now", lambda: clock["t"])
    env.token_status = 400                                    # 垃圾 code：LINE 會拒
    hs = [_login(client, make_user, "ins%d" % i) for i in range(3)]
    victim = _login(client, make_user, "victim")
    sent = 0
    for sec in range(0, 60, 3):                               # 每 3 秒每人一個（＝真實 START_INTERVAL）
        clock["t"] = 1000.0 + sec
        for h in hs:
            st, _ = _begin(client, h, env)
            _cb(client, state=st, code="garbage")
            sent += 1
    assert sent == 60
    token_calls = [c for c in env.calls if c[0].endswith("/token")]
    assert len(token_calls) == 60                            # 60 次真的打到 LINE
    st, _ = _begin(client, victim, env)
    r = _cb(client, state=st, code="legit-code")
    assert r.status_code == 429, "受害者被全站額度擋掉（RED 預期）"
    assert not client.get("/api/line/binding", headers=victim).json()["linked"]

# ── P4 同一 state 並發兩次 ─────────────────────────────────────────────
def test_p4_concurrent_same_state_only_one_wins(client, make_user, env):
    """GREEN：兩條執行緒同時用同一 state 打 complete()，只有一條通過（pop 在鎖內）；LINE 端只被呼叫一次 token。"""
    from helpers import line_login as ll
    hdr = _login(client, make_user, "race_amy")
    st, _ = _begin(client, hdr, env)
    results, gate = [], threading.Barrier(2)

    def run():
        gate.wait()
        results.append(ll.complete("ok-code", st)[0])

    ts = [threading.Thread(target=run) for _ in range(2)]
    [t.start() for t in ts]
    [t.join() for t in ts]
    assert sorted(results) == ["invalid", "linked"], results
    assert len([c for c in env.calls if c[0].endswith("/token")]) == 1

# ── P7 讀取失敗被當成『沒有綁定』→ 整份綁定表被覆寫 ──────────────────────
def test_p7_transient_read_failure_wipes_every_other_users_binding(client, make_user, env, monkeypatch):
    """RED（建議級）：line_links 是**單一 JSON 值**，link_user／unlink_user／set_confirmed 都是 讀→改→寫。
    `_get_setting` 對任何例外回預設值 {}，於是一次暫時性讀取失敗（locked／I/O）就讓 `links={}`，
    寫回時把『所有其他人』的綁定一起抹掉，且無任何錯誤。"""
    from helpers import line_notify as ln
    import helpers.settings as hs
    ln.link_user("alice", "U" + "1" * 32)
    ln.link_user("bob", "U" + "2" * 32)
    assert ln.is_linked("alice") and ln.is_linked("bob")
    real_get = hs.get_db
    state = {"n": 0}

    class _Flaky:
        def __init__(self, c):
            self._c = c

        def execute(self, *a, **k):
            state["n"] += 1
            if state["n"] == 1:               # 第一次 SELECT 暫時失敗（與 get_db 成功連線之後的 busy／I/O 同一個落點）
                raise __import__("sqlite3").OperationalError("database is locked")
            return self._c.execute(*a, **k)

        def __getattr__(self, n):
            return getattr(self._c, n)

    monkeypatch.setattr(hs, "get_db", lambda: _Flaky(real_get()))
    ln.link_user("carol", "U" + "3" * 32)           # 讀取第一次失敗 → 以為沒有任何綁定
    monkeypatch.setattr(hs, "get_db", real_get)
    assert ln.is_linked("carol")
    assert ln.is_linked("alice") is False and ln.is_linked("bob") is False   # 被抹掉（RED 預期）
