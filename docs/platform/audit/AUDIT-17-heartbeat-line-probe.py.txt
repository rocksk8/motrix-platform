import json

TOKEN = "tok-SECRET-0123456789"
EV = "approval_review_edit"


def _uid(n):
    return "U" + ("%032x" % n)


class _Fake:
    def __init__(self):
        self.calls = []

    def __call__(self, method, path, token, payload=None):
        self.calls.append((method, path, payload))
        return 200, {}

    def targets(self):
        out = set()
        for _m, p, pl in self.calls:
            if p.endswith("/push"):
                out.add(pl["to"])
            elif p.endswith("/multicast"):
                out.update(pl["to"])
        return out


class _H:
    def wait(self):
        return "sent"

def _wait(fake, n=1, t=3.0):
    end = time.time() + t
    while time.time() < end and len(fake.calls) < n:
        time.sleep(0.05)
    time.sleep(0.2)


def test_P1_bound_user_without_email_never_gets_line(env):
    en, L, fake, u = env
    en.send_registered(EV, title="t", rows=[("a", "b")], usernames=["amy", "noemail"], wait=False)
    _wait(fake)
    print("P1 targets:", sorted(fake.targets()), "noemail uid:", u["noemail"])
    assert u["noemail"] not in fake.targets() and u["amy"] in fake.targets()


def test_P2_email_mute_silently_kills_line_while_prefs_view_says_enabled(env):
    en, L, fake, u = env
    from db import get_db
    from helpers import line_recipients as lr
    conn = get_db()
    conn.execute("UPDATE users SET notification_muted=? WHERE username='muted'", (json.dumps([EV]),))
    conn.commit()
    conn.close()
    en.send_registered(EV, title="t", rows=[], usernames=["amy", "muted"], wait=False)
    _wait(fake)
    view = {e["key"]: e["enabled"] for e in lr.prefs_view("muted", "sales")}
    print("P2 targets:", sorted(fake.targets()), "prefs_view(muted):", view)
    assert u["muted"] not in fake.targets()
    assert view.get(EV) is True          # UI 仍顯示「要收」


def test_P3_inactive_bound_user_fast_path_vs_restricted_path(env):

def test_P4_superadmin_only_override_still_lists_event_for_engineer(env):
    en, L, fake, u = env
    from helpers import line_recipients as lr
    from helpers import mail_types as mt

def test_P6_all_roles_unchecked_event_silently_dead(env):
    en, L, fake, u = env
    L.set_event_roles(EV, [])
    en.send_registered(EV, title="t", rows=[], usernames=["amy", "boss"], wait=False)

def test_P7_heartbeat_import_leaks_open_log_handle(monkeypatch, tmp_path):
    import importlib
    import logging
    from core import paths
    logs = tmp_path / "joblogs"
    monkeypatch.setattr(paths, "LOGS_DIR", str(logs))
    monkeypatch.setattr(paths, "HEARTBEAT_CONFIG", str(tmp_path / "hbj.json"))
    sys.modules.pop("heartbeat_job", None)
    importlib.import_module("heartbeat_job")
    sys.modules.pop("heartbeat_job", None)
    root_handlers = [h for h in logging.getLogger().handlers if getattr(h, "baseFilename", "").startswith(str(logs))]
    log = logs / "heartbeat_job.log"
    try:
        log.unlink()
        removable = True
    except OSError as e:
        removable = False
        print("P7 unlink error:", e)
    print("P7 attached_to_root:", len(root_handlers), " log_exists_before_unlink: True  removable:", removable)
