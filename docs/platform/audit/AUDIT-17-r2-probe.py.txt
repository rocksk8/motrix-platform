# -*- coding: utf-8 -*-
"""AUDIT-17 r2 delta probes（0bbce3c73 的 token 指紋快取）。用法：複製成 backend/tests/test_zz_audit17r2_probe.py 跑完即刪。"""
from datetime import timedelta

import pytest

import importlib.util, pathlib
_sp = importlib.util.spec_from_file_location("_tlq", pathlib.Path(__file__).with_name("test_line_quota_2026_11_12.py"))
_m = importlib.util.module_from_spec(_sp); _sp.loader.exec_module(_m)
q, _Line, _Timers, T0, TOKEN = _m.q, _m._Line, _m._Timers, _m.T0, _m.TOKEN


def _swap_token(tok):
    from helpers.settings import _get_setting, _set_setting
    c = _get_setting("line_notify", {})
    c["channel_access_token"] = tok
    _set_setting("line_notify", c)


def test_P8_alerted_flags_survive_token_swap(q):
    """A 帳號用到 85%（已警示 80）→ 換成 B 帳號（78%）：alerted 帶過去；B 之後真的到 80% 不會寄。"""
    q.fake.used, q.fake.limit = 170, 200
    v = q.check_and_alert(now=T0)
    assert v["alerted"] is True and len(q.mails) == 1
    _swap_token(TOKEN + "B")
    q.fake.used = 156                                   # B 帳號 78%
    q.refresh(T0 + timedelta(seconds=5))
    from helpers.settings import _get_setting
    st = _get_setting(q.STATE_KEY, {})
    print("P8 state after swap:", st["alerted"])
    q.fake.used = 160                                   # B 到 80%
    q.check_and_alert(now=T0 + timedelta(seconds=400))
    print("P8 mails total:", len(q.mails), "(B 首次跨 80% 應該要有第 2 封)")
    assert len(q.mails) == 2, "B 帳號首次跨 80% 的警示被 A 帳號的已警示旗標吃掉"


def test_P9_failed_read_after_swap_shows_old_account_numbers(q):
    q.fake.used = 160
    assert q.snapshot(now=T0)["used"] == 160
    _swap_token(TOKEN + "B")
    q.fake.status = 401                                 # 新 token 讀不到
    v = q.snapshot(now=T0 + timedelta(seconds=5))
    print("P9 view:", {k: v[k] for k in ("ok", "used", "limit", "percent", "level", "error")})
    assert v["ok"] is False and v["used"] is None, "讀不到新帳號用量時不該帶出舊帳號的 used/limit"


@pytest.mark.line_quota_real_timer
def test_P10_after_push_delay_after_swap(q, monkeypatch):
    timers = _Timers()
    monkeypatch.setattr(q.threading, "Timer", timers)
    q.refresh(T0)
    _swap_token(TOKEN + "B")
    q.after_push(T0 + timedelta(seconds=10))
    print("P10 delay:", timers.made[-1].delay)
    assert timers.made[-1].delay == q._SETTLE_SECONDS


def test_P11_token_swapped_during_read_is_not_cached_as_new_token(q):
    """讀的途中換 token：這次結果不能算新 token 的快取（可殺 M5）。"""
    from helpers import line_notify as L
    real = q.fake
    state = {"n": 0}

    def swapping(method, path, token, payload=None):
        if state["n"] == 0:
            state["n"] = 1
            _swap_token(TOKEN + "B")                    # 第一個 GET 進行中，管理員換了 token
        return real(method, path, token, payload)
    import helpers.line_notify as ln
    ln._http = swapping
    try:
        q.snapshot(now=T0)
        n = len(real.calls)
        q.snapshot(now=T0 + timedelta(seconds=5))
        assert len(real.calls) > n, "換 token 之後必須重讀"
    finally:
        ln._http = real
