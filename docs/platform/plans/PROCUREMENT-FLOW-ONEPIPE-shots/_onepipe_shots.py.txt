def _sql(sql, args=()):
    from db import get_db
    conn = get_db()
    try:
        cur = conn.execute(sql, args)
        conn.commit()
        return cur.lastrowid
    finally:
        conn.close()


def _cols(table):
    from db import get_db
    conn = get_db()
    try:
        return {r[1] for r in conn.execute(f"PRAGMA table_info({table})").fetchall()}
    finally:
        conn.close()


def _insert(table, row):
    cols = _cols(table)
    row = {k: v for k, v in row.items() if k in cols}
    _sql(f"INSERT INTO {table} ({','.join(row)}) VALUES ({','.join('?' * len(row))})", tuple(row.values()))

def _run_steps(page, base, steps, shots, name):
    """steps: [(標題, 動作 callable)]；每步截圖，記錄點擊／換頁次數。"""
    for i, (title, act) in enumerate(steps, 1):
        act()
        page.wait_for_timeout(900)
        fn = f"{name}-{i:02d}.png"
        page.screenshot(path=str(OUT / fn), full_page=False)
        shots.append({"file": fn, "walk": name, "title": title, "url": page.url.replace(base, "")})
